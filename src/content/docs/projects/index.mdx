document.addEventListener('DOMContentLoaded', function(){
  var ctrls = document.getElementById('proj-controls');
  if(!ctrls) return;
  ctrls.querySelectorAll('.dropdown-toggle').forEach(function(btn){
    btn.onclick = function(e){ e.stopPropagation(); var m=this.nextElementSibling; m.style.display=m.style.display==='block'?'none':'block'; };
  });
  ctrls.querySelectorAll('.dropdown-menu').forEach(function(m){
    m.onclick = function(e){ e.stopPropagation(); };
  });
  ctrls.querySelectorAll('input[type=checkbox]').forEach(function(cb){
    cb.onchange = function(){ window.applyProjFilters(); };
  });
  document.addEventListener('click', function(e){ if(!e.target.closest('.dropdown')) document.querySelectorAll('#proj-controls .dropdown-menu').forEach(function(m){m.style.display='none';}); });
});
window.applyProjFilters = function() {
  var cards = document.querySelectorAll('#proj-grid .project-card');
  var groups = document.querySelectorAll('#proj-controls .dropdown-menu');
  cards.forEach(function(card){
    var tags = card.getAttribute('data-tags').split(' ');
    var show = true;
    groups.forEach(function(group){
      var checked = [];
      group.querySelectorAll('input[type=checkbox]').forEach(function(c){ if(c.checked) checked.push(c.value); });
      if(checked.length === 0) return;
      var match = false;
      for(var i=0;i<tags.length;i++){ if(checked.indexOf(tags[i])!==-1){match=true;break;} }
      if(!match) show = false;
    });
    card.style.display = show ? '' : 'none';
  });
};
`}</script>

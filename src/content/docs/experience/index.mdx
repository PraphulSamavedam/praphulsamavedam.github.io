document.addEventListener('DOMContentLoaded', function(){
  var ctrls = document.getElementById('exp-controls');
  if(!ctrls) return;
  ctrls.querySelectorAll('.dropdown-toggle').forEach(function(btn){
    btn.onclick = function(e){ e.stopPropagation(); var m=this.nextElementSibling; m.style.display=m.style.display==='block'?'none':'block'; };
  });
  ctrls.querySelectorAll('.dropdown-menu').forEach(function(m){
    m.onclick = function(e){ e.stopPropagation(); };
  });
  ctrls.querySelectorAll('input[type=checkbox]').forEach(function(cb){
    cb.onchange = function(){ window.applyExpFilters(); };
  });
  var sel = ctrls.querySelector('.sort-select');
  if(sel) sel.onchange = function(){ window.sortExp(this.value); };
  document.addEventListener('click', function(e){ if(!e.target.closest('.dropdown')) document.querySelectorAll('.dropdown-menu').forEach(function(m){m.style.display='none';}); });
});
window.applyExpFilters = function() {
  var cards = document.querySelectorAll('#exp-grid .experience-card');
  var groups = document.querySelectorAll('#exp-controls .dropdown-menu');
  cards.forEach(function(card){
    var tags = card.getAttribute('data-tags').split(' ');
    var show = true;
    groups.forEach(function(group){
      var checked = [];
      group.querySelectorAll('input[type=checkbox]').forEach(function(c){ if(c.checked) checked.push(c.value); });
      if(checked.length === 0) return;
      var match = false;
      for(var i=0;i<tags.length;i++){ if(checked.indexOf(tags[i])!==-1){match=true;break;} }
      if(!match) show = false;
    });
    card.style.display = show ? '' : 'none';
  });
};
window.sortExp = function(val) {
  var grid = document.getElementById('exp-grid');
  var cards = Array.from(grid.querySelectorAll('.experience-card'));
  var p = val.split('-'), key = p[0], dir = p[1];
  cards.sort(function(a,b){
    var va = key==='duration' ? Number(a.getAttribute('data-months')) : a.getAttribute('data-start');
    var vb = key==='duration' ? Number(b.getAttribute('data-months')) : b.getAttribute('data-start');
    return dir==='desc' ? (vb>va?1:-1) : (va>vb?1:-1);
  });
  cards.forEach(function(c){ grid.appendChild(c); });
};
`}</script>

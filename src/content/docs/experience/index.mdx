<script>{`
(function () {
  const tl = document.querySelector('.experience-timeline');
  if (!tl) return;
  let rows = Array.from(tl.querySelectorAll('.lt-row'));
  if (!rows.length) return;
  tl.classList.add('lt-scroll-ready');

  const setActive = (el) => {
    rows.forEach((r) => r.classList.toggle('lt-active', r === el));
  };

  const pickClosest = () => {
    const mid = window.innerHeight / 2;
    let best = null, bestDist = Infinity;
    for (const r of rows) {
      const box = r.getBoundingClientRect();
      const dist = Math.abs(box.top + box.height / 2 - mid);
      if (dist < bestDist) { bestDist = dist; best = r; }
    }
    if (best) setActive(best);
  };

  let ticking = false;
  let hovering = false;
  const onScroll = () => {
    if (hovering) return;
    if (ticking) return;
    ticking = true;
    requestAnimationFrame(() => { if (!hovering) pickClosest(); ticking = false; });
  };

  rows.forEach((r) => {
    r.addEventListener('mouseenter', () => { hovering = true; setActive(r); });
  });
  tl.addEventListener('mouseleave', () => { hovering = false; pickClosest(); });

  // Whole-card navigation: a card with a data-href navigates to it on click.
  // The inner title <a> keeps its native behavior (and keyboard focus).
  rows.forEach((r) => {
    const card = r.querySelector('.lt-card');
    if (!card || !card.dataset.href) return;
    card.classList.add('lt-clickable');
    card.addEventListener('click', (e) => {
      if (e.target.closest('a')) return;                       // let real links work
      window.location.href = card.dataset.href;
    });
  });

  // Sort control: re-append rows in the chosen order. The spine's :last-child
  // shortening follows DOM order automatically, so the rail stays correct.
  const sortBy = (val) => {
    const [key, dir] = val.split('-');
    rows.sort((a, b) => {
      const va = key === 'duration' ? Number(a.dataset.months) : a.dataset.start;
      const vb = key === 'duration' ? Number(b.dataset.months) : b.dataset.start;
      if (va === vb) return 0;
      return dir === 'desc' ? (vb > va ? 1 : -1) : (va > vb ? 1 : -1);
    });
    rows.forEach((r) => tl.appendChild(r));
    setActive(rows[0]);
  };
  const sel = document.querySelector('#exp-controls .sort-select');
  if (sel) sel.addEventListener('change', function () { sortBy(this.value); });

  window.addEventListener('scroll', onScroll, { passive: true });
  window.addEventListener('resize', onScroll, { passive: true });
  setActive(rows[0]);
})();
`}</script>
